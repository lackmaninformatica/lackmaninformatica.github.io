#requires -RunAsAdministrator

$ErrorActionPreference = "SilentlyContinue"

Write-Host "========================================="
Write-Host " REMOCAO COMPLETA DO MICROSOFT ONEDRIVE"
Write-Host "========================================="

# Encerra processos do OneDrive
Write-Host "`nEncerrando processos..."

$processos = @(
    "OneDrive",
    "OneDriveSetup",
    "FileCoAuth",
    "Microsoft.SharePoint"
)

foreach ($processo in $processos) {
    Stop-Process -Name $processo -Force -ErrorAction SilentlyContinue
}

taskkill.exe /F /IM OneDrive.exe /T 2>$null
taskkill.exe /F /IM OneDriveSetup.exe /T 2>$null

# Localiza o desinstalador nativo
$desinstaladores = @(
    "$env:SystemRoot\System32\OneDriveSetup.exe",
    "$env:SystemRoot\SysWOW64\OneDriveSetup.exe",
    "$env:LOCALAPPDATA\Microsoft\OneDrive\OneDriveSetup.exe",
    "$env:ProgramFiles\Microsoft OneDrive\OneDriveSetup.exe",
    "${env:ProgramFiles(x86)}\Microsoft OneDrive\OneDriveSetup.exe"
) | Select-Object -Unique

foreach ($desinstalador in $desinstaladores) {
    if (Test-Path $desinstalador) {
        Write-Host "Executando desinstalador: $desinstalador"

        Start-Process `
            -FilePath $desinstalador `
            -ArgumentList "/uninstall" `
            -Wait `
            -ErrorAction SilentlyContinue
    }
}

# Tenta desinstalar por Winget
if (Get-Command winget.exe -ErrorAction SilentlyContinue) {
    Write-Host "Tentando remover pelo Winget..."

    winget.exe uninstall `
        --id Microsoft.OneDrive `
        --silent `
        --accept-source-agreements `
        --disable-interactivity
}

# Tenta remover pacotes AppX relacionados
Write-Host "Removendo pacotes AppX relacionados..."

Get-AppxPackage -AllUsers |
    Where-Object {
        $_.Name -match "OneDrive"
    } |
    ForEach-Object {
        Write-Host "Removendo pacote: $($_.Name)"

        Remove-AppxPackage `
            -Package $_.PackageFullName `
            -AllUsers `
            -ErrorAction SilentlyContinue
    }

Get-AppxProvisionedPackage -Online |
    Where-Object {
        $_.DisplayName -match "OneDrive"
    } |
    ForEach-Object {
        Write-Host "Removendo provisionamento: $($_.DisplayName)"

        Remove-AppxProvisionedPackage `
            -Online `
            -PackageName $_.PackageName `
            -AllUsers `
            -ErrorAction SilentlyContinue
    }

Start-Sleep -Seconds 3

# Remove tarefas agendadas do OneDrive
Write-Host "Removendo tarefas agendadas..."

Get-ScheduledTask -ErrorAction SilentlyContinue |
    Where-Object {
        $_.TaskName -match "OneDrive" -or
        $_.TaskPath -match "OneDrive"
    } |
    ForEach-Object {
        Write-Host "Removendo tarefa: $($_.TaskPath)$($_.TaskName)"

        Unregister-ScheduledTask `
            -TaskName $_.TaskName `
            -TaskPath $_.TaskPath `
            -Confirm:$false `
            -ErrorAction SilentlyContinue
    }

# Remove inicialização automática
Write-Host "Removendo inicialização automática..."

$chavesRun = @(
    "HKCU:\Software\Microsoft\Windows\CurrentVersion\Run",
    "HKLM:\Software\Microsoft\Windows\CurrentVersion\Run",
    "HKLM:\Software\WOW6432Node\Microsoft\Windows\CurrentVersion\Run"
)

foreach ($chave in $chavesRun) {
    if (Test-Path $chave) {
        Remove-ItemProperty `
            -Path $chave `
            -Name "OneDrive" `
            -Force `
            -ErrorAction SilentlyContinue
    }
}

# Obtém perfis locais
$perfis = Get-CimInstance Win32_UserProfile |
    Where-Object {
        -not $_.Special -and
        $_.LocalPath -like "C:\Users\*"
    }

# Remove pastas do usuário atual e de outros perfis
$pastas = @(
    "$env:LOCALAPPDATA\Microsoft\OneDrive",
    "$env:LOCALAPPDATA\OneDrive",
    "$env:APPDATA\Microsoft\OneDrive",
    "$env:PROGRAMDATA\Microsoft OneDrive",
    "$env:ProgramFiles\Microsoft OneDrive",
    "${env:ProgramFiles(x86)}\Microsoft OneDrive",
    "$env:SystemDrive\OneDriveTemp",
    "$env:TEMP\OneDrive*"
)

foreach ($perfil in $perfis) {
    $pastas += @(
        "$($perfil.LocalPath)\AppData\Local\Microsoft\OneDrive",
        "$($perfil.LocalPath)\AppData\Local\OneDrive",
        "$($perfil.LocalPath)\AppData\Roaming\Microsoft\OneDrive",
        "$($perfil.LocalPath)\AppData\Local\Microsoft\OneDriveSetup",
        "$($perfil.LocalPath)\AppData\Local\Temp\OneDrive*"
    )
}

foreach ($pasta in ($pastas | Select-Object -Unique)) {
    Get-Item $pasta -Force -ErrorAction SilentlyContinue |
        ForEach-Object {
            Write-Host "Removendo: $($_.FullName)"

            takeown.exe /F "$($_.FullName)" /R /D Y 2>$null | Out-Null
            icacls.exe "$($_.FullName)" /grant "*S-1-5-32-544:F" /T /C 2>$null | Out-Null

            Remove-Item `
                -LiteralPath $_.FullName `
                -Recurse `
                -Force `
                -ErrorAction SilentlyContinue
        }
}

# Remove a pasta OneDrive do usuário atual
# Atenção: isto remove arquivos locais que ainda estiverem nessa pasta
$pastasSincronizadas = @(
    "$env:USERPROFILE\OneDrive",
    "$env:USERPROFILE\OneDrive - Personal",
    "$env:USERPROFILE\OneDrive - Pessoal"
)

foreach ($pasta in $pastasSincronizadas) {
    if (Test-Path $pasta) {
        Write-Warning "Removendo pasta local sincronizada: $pasta"

        takeown.exe /F "$pasta" /R /D Y 2>$null | Out-Null
        icacls.exe "$pasta" /grant "*S-1-5-32-544:F" /T /C 2>$null | Out-Null

        Remove-Item $pasta -Recurse -Force -ErrorAction SilentlyContinue
    }
}

# Remove atalhos
Write-Host "Removendo atalhos..."

$atalhos = @(
    "$env:PUBLIC\Desktop\OneDrive.lnk",
    "$env:USERPROFILE\Desktop\OneDrive.lnk",
    "$env:APPDATA\Microsoft\Windows\Start Menu\Programs\OneDrive.lnk",
    "$env:ProgramData\Microsoft\Windows\Start Menu\Programs\OneDrive.lnk"
)

foreach ($atalho in $atalhos) {
    Remove-Item $atalho -Force -ErrorAction SilentlyContinue
}

# Remove chaves de registro residuais
Write-Host "Removendo registros residuais..."

$registros = @(
    "HKCU:\Software\Microsoft\OneDrive",
    "HKCU:\Software\Microsoft\OneDriveSetup",
    "HKCU:\Software\SyncEngines\Providers\OneDrive",
    "HKCU:\Software\Microsoft\Windows\CurrentVersion\Explorer\SyncRootManager\OneDrive",
    "HKLM:\Software\Microsoft\OneDrive",
    "HKLM:\Software\WOW6432Node\Microsoft\OneDrive"
)

foreach ($registro in $registros) {
    if (Test-Path $registro) {
        Write-Host "Removendo registro: $registro"

        Remove-Item `
            -Path $registro `
            -Recurse `
            -Force `
            -ErrorAction SilentlyContinue
    }
}

# Remove entradas do OneDrive no painel lateral do Explorador
$namespaceKeys = @(
    "Registry::HKEY_CLASSES_ROOT\CLSID\{018D5C66-4533-4307-9B53-224DE2ED1FE6}",
    "Registry::HKEY_CLASSES_ROOT\WOW6432Node\CLSID\{018D5C66-4533-4307-9B53-224DE2ED1FE6}",
    "HKCU:\Software\Classes\CLSID\{018D5C66-4533-4307-9B53-224DE2ED1FE6}"
)

foreach ($chave in $namespaceKeys) {
    if (Test-Path $chave) {
        Set-ItemProperty `
            -Path $chave `
            -Name "System.IsPinnedToNameSpaceTree" `
            -Value 0 `
            -Type DWord `
            -Force `
            -ErrorAction SilentlyContinue
    }
}

# Bloqueia temporariamente a integração do OneDrive
# Remova esta política caso pretenda reinstalar no futuro
$politica = "HKLM:\SOFTWARE\Policies\Microsoft\Windows\OneDrive"

if (-not (Test-Path $politica)) {
    New-Item -Path $politica -Force | Out-Null
}

New-ItemProperty `
    -Path $politica `
    -Name "DisableFileSyncNGSC" `
    -PropertyType DWord `
    -Value 1 `
    -Force | Out-Null

# Reinicia o Explorador do Windows
Write-Host "Reiniciando o Explorador do Windows..."

Stop-Process -Name explorer -Force -ErrorAction SilentlyContinue
Start-Sleep -Seconds 2
Start-Process explorer.exe

# Verificação final
Write-Host "`n========================================="
Write-Host " VERIFICACAO FINAL"
Write-Host "========================================="

$processoRestante = Get-Process -Name OneDrive -ErrorAction SilentlyContinue

$instalacoesRestantes = @(
    "$env:LOCALAPPDATA\Microsoft\OneDrive",
    "$env:ProgramFiles\Microsoft OneDrive",
    "${env:ProgramFiles(x86)}\Microsoft OneDrive"
) | Where-Object {
    Test-Path $_
}

if (-not $processoRestante -and -not $instalacoesRestantes) {
    Write-Host "O OneDrive foi removido."
}
else {
    if ($processoRestante) {
        Write-Warning "O processo OneDrive ainda está ativo."
    }

    if ($instalacoesRestantes) {
        Write-Warning "Ainda existem pastas:"
        $instalacoesRestantes
    }
}

Write-Host "`nReinicie o computador para concluir a limpeza."
