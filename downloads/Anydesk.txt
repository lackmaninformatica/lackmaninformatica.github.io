# Remocao completa do AnyDesk - limpeza operacional segura
# Nao apaga logs/eventos do Windows

$ErrorActionPreference = "SilentlyContinue"

Write-Host "=== REMOCAO COMPLETA DO ANYDESK ===" -ForegroundColor Cyan

# 1. Encerrar processos
Write-Host "Encerrando processos do AnyDesk..."
Get-Process | Where-Object {
    $_.ProcessName -match "AnyDesk"
} | Stop-Process -Force

Start-Sleep -Seconds 2

# 2. Parar e remover servicos
Write-Host "Removendo servicos..."
$services = Get-Service | Where-Object {
    $_.Name -match "AnyDesk" -or $_.DisplayName -match "AnyDesk"
}

foreach ($svc in $services) {
    Stop-Service -Name $svc.Name -Force
    sc.exe delete $svc.Name | Out-Null
}

# 3. Remover tarefas agendadas relacionadas
Write-Host "Removendo tarefas agendadas..."
Get-ScheduledTask | Where-Object {
    $_.TaskName -match "AnyDesk" -or $_.TaskPath -match "AnyDesk"
} | ForEach-Object {
    Unregister-ScheduledTask -TaskName $_.TaskName -TaskPath $_.TaskPath -Confirm:$false
}

# 4. Tentar desinstalacao normal pelo registro
Write-Host "Procurando desinstaladores..."
$uninstallKeys = @(
    "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall\*",
    "HKLM:\SOFTWARE\WOW6432Node\Microsoft\Windows\CurrentVersion\Uninstall\*",
    "HKCU:\SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall\*"
)

foreach ($key in $uninstallKeys) {
    Get-ItemProperty $key | Where-Object {
        $_.DisplayName -match "AnyDesk"
    } | ForEach-Object {
        Write-Host "Encontrado: $($_.DisplayName)"
        
        if ($_.QuietUninstallString) {
            Write-Host "Executando desinstalacao silenciosa..."
            Start-Process "cmd.exe" -ArgumentList "/c $($_.QuietUninstallString)" -Wait
        }
        elseif ($_.UninstallString) {
            Write-Host "Executando desinstalador..."
            $cmd = $_.UninstallString
            
            if ($cmd -match "msiexec") {
                $cmd = $cmd -replace "/I", "/X"
                Start-Process "cmd.exe" -ArgumentList "/c $cmd /qn /norestart" -Wait
            } else {
                Start-Process "cmd.exe" -ArgumentList "/c `"$cmd`" /S" -Wait
            }
        }
    }
}

Start-Sleep -Seconds 3

# 5. Remover instaladores conhecidos
Write-Host "Removendo instalacoes conhecidas..."
$possibleInstallers = @(
    "$env:ProgramFiles\AnyDesk\AnyDesk.exe",
    "${env:ProgramFiles(x86)}\AnyDesk\AnyDesk.exe",
    "$env:LOCALAPPDATA\Programs\AnyDesk\AnyDesk.exe"
)

foreach ($exe in $possibleInstallers) {
    if (Test-Path $exe) {
        Start-Process $exe -ArgumentList "--remove" -Wait
        Start-Sleep -Seconds 2
    }
}

# 6. Encerrar novamente processos restantes
Get-Process | Where-Object {
    $_.ProcessName -match "AnyDesk"
} | Stop-Process -Force

# 7. Remover pastas do sistema e usuario
Write-Host "Removendo pastas..."
$folders = @(
    "$env:ProgramFiles\AnyDesk",
    "${env:ProgramFiles(x86)}\AnyDesk",
    "$env:ProgramData\AnyDesk",
    "$env:APPDATA\AnyDesk",
    "$env:LOCALAPPDATA\AnyDesk",
    "$env:LOCALAPPDATA\Programs\AnyDesk",
    "$env:TEMP\AnyDesk"
)

foreach ($folder in $folders) {
    if (Test-Path $folder) {
        Write-Host "Removendo: $folder"
        Remove-Item $folder -Recurse -Force
    }
}

# 8. Remover AnyDesk de todos os perfis de usuario
Write-Host "Limpando perfis de usuarios..."
$users = Get-ChildItem "C:\Users" -Directory | Where-Object {
    $_.Name -notin @("Public", "Default", "Default User", "All Users")
}

foreach ($user in $users) {
    $userPaths = @(
        "$($user.FullName)\AppData\Roaming\AnyDesk",
        "$($user.FullName)\AppData\Local\AnyDesk",
        "$($user.FullName)\AppData\Local\Programs\AnyDesk",
        "$($user.FullName)\Desktop\AnyDesk.lnk",
        "$($user.FullName)\AppData\Roaming\Microsoft\Windows\Start Menu\Programs\AnyDesk.lnk",
        "$($user.FullName)\AppData\Roaming\Microsoft\Windows\Start Menu\Programs\AnyDesk"
    )

    foreach ($path in $userPaths) {
        if (Test-Path $path) {
            Write-Host "Removendo: $path"
            Remove-Item $path -Recurse -Force
        }
    }
}

# 9. Remover atalhos comuns
Write-Host "Removendo atalhos..."
$shortcuts = @(
    "$env:PUBLIC\Desktop\AnyDesk.lnk",
    "$env:ProgramData\Microsoft\Windows\Start Menu\Programs\AnyDesk.lnk",
    "$env:ProgramData\Microsoft\Windows\Start Menu\Programs\AnyDesk"
)

foreach ($shortcut in $shortcuts) {
    if (Test-Path $shortcut) {
        Remove-Item $shortcut -Recurse -Force
    }
}

# 10. Remover entradas comuns de inicializacao
Write-Host "Limpando inicializacao..."
$runKeys = @(
    "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Run",
    "HKLM:\SOFTWARE\WOW6432Node\Microsoft\Windows\CurrentVersion\Run",
    "HKCU:\SOFTWARE\Microsoft\Windows\CurrentVersion\Run"
)

foreach ($rk in $runKeys) {
    if (Test-Path $rk) {
        Get-ItemProperty $rk | ForEach-Object {
            $_.PSObject.Properties | Where-Object {
                $_.Name -match "AnyDesk" -or $_.Value -match "AnyDesk"
            } | ForEach-Object {
                Remove-ItemProperty -Path $rk -Name $_.Name -Force
            }
        }
    }
}

# 11. Remover chaves de registro do AnyDesk
Write-Host "Removendo registros do AnyDesk..."
$regPaths = @(
    "HKLM:\SOFTWARE\AnyDesk",
    "HKLM:\SOFTWARE\WOW6432Node\AnyDesk",
    "HKCU:\SOFTWARE\AnyDesk"
)

foreach ($reg in $regPaths) {
    if (Test-Path $reg) {
        Remove-Item $reg -Recurse -Force
    }
}

# 12. Remover entradas de uninstall restantes
Write-Host "Limpando entradas de desinstalacao..."
foreach ($key in $uninstallKeys) {
    Get-ItemProperty $key | Where-Object {
        $_.DisplayName -match "AnyDesk"
    } | ForEach-Object {
        if ($_.PSPath) {
            Remove-Item $_.PSPath -Recurse -Force
        }
    }
}

# 13. Remover regras de firewall do AnyDesk
Write-Host "Removendo regras de firewall..."
Get-NetFirewallRule | Where-Object {
    $_.DisplayName -match "AnyDesk"
} | Remove-NetFirewallRule

# 14. Remover arquivos temporarios comuns relacionados
Write-Host "Limpando temporarios relacionados..."
Get-ChildItem $env:TEMP -Recurse -Force | Where-Object {
    $_.Name -match "AnyDesk"
} | Remove-Item -Recurse -Force

Get-ChildItem "C:\Windows\Temp" -Recurse -Force | Where-Object {
    $_.Name -match "AnyDesk"
} | Remove-Item -Recurse -Force

# 15. Conferencia final
Write-Host ""
Write-Host "=== VERIFICACAO FINAL ===" -ForegroundColor Yellow

$foundProcess = Get-Process | Where-Object { $_.ProcessName -match "AnyDesk" }
$foundService = Get-Service | Where-Object { $_.Name -match "AnyDesk" -or $_.DisplayName -match "AnyDesk" }
$foundFolders = $folders | Where-Object { Test-Path $_ }

if (!$foundProcess -and !$foundService -and !$foundFolders) {
    Write-Host "AnyDesk removido operacionalmente com sucesso." -ForegroundColor Green
} else {
    Write-Host "Ainda foram encontrados alguns itens:" -ForegroundColor Red

    if ($foundProcess) {
        Write-Host "Processos:"
        $foundProcess | Select-Object ProcessName, Id
    }

    if ($foundService) {
        Write-Host "Servicos:"
        $foundService | Select-Object Name, DisplayName, Status
    }

    if ($foundFolders) {
        Write-Host "Pastas:"
        $foundFolders
    }
}

Write-Host ""
Write-Host "Recomendado reiniciar o computador agora." -ForegroundColor Cyan

$restart = Read-Host "Deseja reiniciar agora? S/N"
if ($restart -match "^[sS]$") {
    Restart-Computer -Force
}
